#include <stdio.h>
#include <stdlib.h>
#include <time.h> 
#include <locale.h>

int main() {
    setlocale(LC_ALL, "Russian");
    int n = 10;
    int a[10];

    srand(time(NULL));

    printf("Сгенерированный массив:\n");
    for (int i = 0; i < n; i++) {
        a[i] = rand() % 100;
        printf("%d ", a[i]);
    }
    printf("\n");

    return 0;
}
