#include <stdio.h>
#include <locale.h>

int main() {
    setlocale(LC_ALL, "Russian");
    int a[] = { 12, 45, 3, 89, 23, 7, 56, 1, 34, 67 };
    int n = 10; 

    int max = a[0];
    int min = a[0];

    for (int i = 1; i < n; i++) {
        if (a[i] > max) {
            max = a[i];
        }
        if (a[i] < min) {
            min = a[i];
        }
    }

    int diff = max - min;

    printf("Массив: ");
    for (int i = 0; i < n; i++) printf("%d ", a[i]);
    printf("\n");

    printf("Максимальный элемент: %d\n", max);
    printf("Минимальный элемент: %d\n", min);
    printf("Разница: %d\n", diff);

    return 0;
}
