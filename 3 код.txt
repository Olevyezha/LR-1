#include <stdio.h>
#include <stdlib.h>
#include <locale.h>

int main() {
setlocale(LC_ALL, "Russian");

int n;
printf("Введите размер массива: ");
scanf_s("%d", &n);

if (n <= 0) {
printf("Размер должен быть положительным числом!\n");
return 1;
}

int* a = (int*)malloc(n * sizeof(int));
if (a == NULL) {
printf("Ошибка выделения памяти!\n");
return 1;
}

printf("Введите элементы массива:\n");
for (int i = 0; i < n; i++) {
printf("a[%d] = ", i);
scanf_s("%d", &a[i]);
}

printf("\nСозданный массив: ");
for (int i = 0; i < n; i++) {
printf("%d ", a[i]);
}	

printf("\n");	
free(a);
return 0;
}
