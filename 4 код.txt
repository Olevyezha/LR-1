#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <locale.h>

int main() {
    setlocale(LC_ALL, "Russian");
    const int ROWS = 3;
    const int COLS = 4;
    int a[ROWS][COLS];

    srand(time(NULL));

    printf("Матрица %dx%d:\n", ROWS, COLS);
    for (int i = 0; i < ROWS; i++) {
        for (int j = 0; j < COLS; j++) {
            a[i][j] = rand() % 10; 
            printf("%d\t", a[i][j]);
        }
        printf("\n");
    }

    printf("\nСуммы по столбцам:\n");
    for (int j = 0; j < COLS; j++) { 
        int sum = 0;
        for (int i = 0; i < ROWS; i++) { 
            sum += a[i][j];
        }
        printf("Столбец %d: %d\n", j + 1, sum);
    }

    return 0;
}
